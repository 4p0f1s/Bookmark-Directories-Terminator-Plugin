# This plugin has no PyPI dependencies.
# All required packages are system-level:
#
#   Python 3.6+
#   PyGObject (GTK 3 bindings)
#
# Install system dependencies:
#
#   Debian / Ubuntu:
#     sudo apt install python3-gi python3-gi-cairo gir1.2-gtk-3.0 terminator
#
#   Fedora / RHEL:
#     sudo dnf install python3-gobject gtk3 terminator
#
#   Arch Linux:
#     sudo pacman -S python-gobject gtk3 terminator
